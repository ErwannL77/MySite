  <h1>Générateur de QR code</h1>
  <input id="url" type="url" placeholder="https://tondomaine.fr" value="https://">
  <button id="generer">Générer</button>

  <img id="apercu" alt="QR code" hidden>
  <a id="telecharger" download="qrcode.png" hidden>Télécharger en PNG</a>

  <script src="https://cdn.jsdelivr.net/npm/qrcode@1.5.3/build/qrcode.min.js"></script>
  <script>
    const apercu = document.getElementById("apercu");
    const lien = document.getElementById("telecharger");

    document.getElementById("generer").addEventListener("click", async () => {
      const url = document.getElementById("url").value.trim();
      if (!url) return;

      const dataUrl = await QRCode.toDataURL(url, {
        width: 1024,                    // grande image, nette à l'impression
        margin: 4,                      // marge blanche, indispensable pour être scanné
        errorCorrectionLevel: "M",
        color: { dark: "#000000", light: "#ffffff" }
      });

      apercu.src = dataUrl;
      apercu.hidden = false;
      lien.href = dataUrl;
      lien.hidden = false;
    });
  </script>
